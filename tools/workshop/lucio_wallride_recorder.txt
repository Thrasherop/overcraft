		8: wallNormal
		9: rideCount
		10: lastEvent
		11: marks
		12: ray
		13: rayStart
		14: rayEnd

		16: wallDist

{

	}

	}
}

			Visible To and String, Default Visibility);

rule("Every tick: the nearest wall around her, then three log lines")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Has Spawned(Event Player) == True;
		Is Alive(Event Player) == True;
	}

	actions
	{
		Event Player.tick = Event Player.tick + 1;
		"The wall: eight level rays at chest height, every 45 degrees round her (not tied to where she looks), 1.5 m each; the nearest hit wins"
		Event Player.rayStart = Position Of(Event Player) + Vector(0, 1, 0);
		Event Player.wallDist = 9;
		Event Player.wallNormal = Vector(0, 0, 0);
		For Player Variable(Event Player, ray, 0, 8, 1);
			Event Player.rayEnd = Event Player.rayStart + Direction From Angles(Event Player.ray * 45, 0) * 1.5;
			Event Player.rayHit = Ray Cast Hit Position(Event Player.rayStart, Event Player.rayEnd, Null, All Players(All Teams), False);
			If(Distance Between(Event Player.rayStart, Event Player.rayHit) < 1.49 && Distance Between(Event Player.rayStart,
				Event Player.rayHit) < Event Player.wallDist);
				Event Player.wallDist = Distance Between(Event Player.rayStart, Event Player.rayHit);
				Event Player.wallNormal = Ray Cast Hit Normal(Event Player.rayStart, Event Player.rayEnd, Null, All Players(All Teams),
					False);
			End;
		End;
		"S slot | T time | wall ground air"
		Log To Inspector(Custom String("S{0} T {1} | {2}", Slot Of(Event Player), Total Time Elapsed, Custom String("{0} {1} {2}",
			Is On Wall(Event Player), Is On Ground(Event Player), Is In Air(Event Player))));
		"S slot | P position | V velocity | I move input"
		Log To Inspector(Custom String("S{0} P {1} | {2}", Slot Of(Event Player), Position Of(Event Player), Custom String("V {0} | I {1}",
			Velocity Of(Event Player), Throttle Of(Event Player))));
		"S slot | B jump crouch | F facing | N wall normal | D wall distance (9: no wall within 1.5 m)"
		Log To Inspector(Custom String("S{0} B {1} | {2}", Slot Of(Event Player), Custom String("{0} {1}", Is Button Held(Event Player,
			Button(Jump)), Is Button Held(Event Player, Button(Crouch))), Custom String("F {0} | N {1} | D {2}", Facing Direction Of(
			Event Player), Event Player.wallNormal, Event Player.wallDist)));
		"S slot | A primary secondary crouching | using ability 1, ability 2 (Amp It Up), ultimate | crossfades so far"
		Log To Inspector(Custom String("S{0} A {1} | {2}", Slot Of(Event Player), Custom String("{0} {1} {2}", Is Button Held(
			Event Player, Button(Primary Fire)), Is Button Held(Event Player, Button(Secondary Fire)), Is Crouching(Event Player)),
			Custom String("U {0} | X {1}", Custom String("{0} {1} {2}", Is Using Ability 1(Event Player), Is Using Ability 2(
			Event Player), Is Using Ultimate(Event Player)), Event Player.crossfades)));
		Wait(0.016, Ignore Condition);
		Loop If Condition Is True;
	}
}

rule("On a wall: a bookmark (no waits, so a quick next wall still gets its own)")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is On Wall(Event Player) == True;
	}

	actions
	{
		Event Player.attachTime = Total Time Elapsed;
		Event Player.attachPos = Position Of(Event Player);
		Event Player.attachVel = Velocity Of(Event Player);
		Event Player.rideCount = Event Player.rideCount + 1;
		Event Player.lastEvent = Custom String("ATTACH {0} | vel {1} | input {2}", Event Player.rideCount, Event Player.attachVel,
			Throttle Of(Event Player));
		Log To Inspector(Custom String("S{0} EVENT ATTACH {1} | {2}", Slot Of(Event Player), Event Player.rideCount, Custom String(
			"t {0} | vel {1} | {2}", Total Time Elapsed, Velocity Of(Event Player), Custom String("pos {0} | face {1} | N {2}",
			Position Of(Event Player), Facing Direction Of(Event Player), Event Player.wallNormal))));
	}
}

rule("Off the wall: a bookmark (no waits)")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is On Wall(Event Player) == False;
		Event Player.attachTime > Event Player.leaveTime;
	}

	actions
	{
		Event Player.leaveTime = Total Time Elapsed;
		Event Player.leaveVel = Velocity Of(Event Player);
		Event Player.lastEvent = Custom String("LEAVE after {0} s | vel {1} | jump on leaving {2}", Round To Integer((Total Time Elapsed
			- Event Player.attachTime) * 100, Down) / 100, Event Player.leaveVel, Is Button Held(Event Player, Button(Jump)));
		Log To Inspector(Custom String("S{0} EVENT LEAVE {1} | {2}", Slot Of(Event Player), Event Player.rideCount, Custom String(
			"t {0} | rode {1} s | {2}", Total Time Elapsed, Total Time Elapsed - Event Player.attachTime, Custom String(
			"vel {0} | rose {1} | jump {2}", Velocity Of(Event Player), Y Component Of(Position Of(Event Player)
			- Event Player.attachPos), Is Button Held(Event Player, Button(Jump))))));
	}
}

rule("Interact marks a moment (say what you just did, and your S number, when you send the log)")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is Button Held(Event Player, Button(Interact)) == True;
	}

	actions
	{
		Event Player.marks = Event Player.marks + 1;
