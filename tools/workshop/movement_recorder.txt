rule("Every tick: the nearest wall around them, then seven log lines")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Has Spawned(Event Player) == True;
		Is Alive(Event Player) == True;
		Is Dummy Bot(Event Player) == False;
	}

	actions
	{
		Event Player.tick = Event Player.tick + 1;
		"The wall: eight level rays at chest height, every 45 degrees round them (not tied to where they look), 1.5 m each; the nearest hit wins"
		Event Player.rayStart = Position Of(Event Player) + Vector(0, 1, 0);
		Event Player.wallDist = 9;
		Event Player.wallNormal = Vector(0, 0, 0);
		For Player Variable(Event Player, ray, 0, 8, 1);
			Event Player.rayEnd = Event Player.rayStart + Direction From Angles(Event Player.ray * 45, 0) * 1.5;
			Event Player.rayHit = Ray Cast Hit Position(Event Player.rayStart, Event Player.rayEnd, Null, All Players(All Teams), False);
			If(Distance Between(Event Player.rayStart, Event Player.rayHit) < 1.49 && Distance Between(Event Player.rayStart,
				Event Player.rayHit) < Event Player.wallDist);
				Event Player.wallDist = Distance Between(Event Player.rayStart, Event Player.rayHit);
				Event Player.wallNormal = Ray Cast Hit Normal(Event Player.rayStart, Event Player.rayEnd, Null, All Players(All Teams),
					False);
			End;
		End;
		"S slot | T time | wall ground air | hero"
		Log To Inspector(Custom String("S{0} T {1} | {2}", Slot Of(Event Player), Total Time Elapsed, Custom String("{0} {1} {2}",
			Is On Wall(Event Player), Is On Ground(Event Player), Custom String("{0} | {1}", Is In Air(Event Player), Hero Of(
			Event Player)))));
		"S slot | P position | V velocity | I move input"
		Log To Inspector(Custom String("S{0} P {1} | {2}", Slot Of(Event Player), Position Of(Event Player), Custom String("V {0} | I {1}",
			Velocity Of(Event Player), Throttle Of(Event Player))));
		"S slot | B jump crouch held | F facing | N wall normal | D wall distance (9: no wall within 1.5 m)"
		Log To Inspector(Custom String("S{0} B {1} | {2}", Slot Of(Event Player), Custom String("{0} {1}", Is Button Held(Event Player,
			Button(Jump)), Is Button Held(Event Player, Button(Crouch))), Custom String("F {0} | N {1} | D {2}", Facing Direction Of(
			Event Player), Event Player.wallNormal, Event Player.wallDist)));
		"S slot | A primary secondary held, crouching | U using ability 1, ability 2, ultimate | X ability 1 presses so far"
		Log To Inspector(Custom String("S{0} A {1} | {2}", Slot Of(Event Player), Custom String("{0} {1} {2}", Is Button Held(
			Event Player, Button(Primary Fire)), Is Button Held(Event Player, Button(Secondary Fire)), Is Crouching(Event Player)),
			Custom String("U {0} | X {1}", Custom String("{0} {1} {2}", Is Using Ability 1(Event Player), Is Using Ability 2(
			Event Player), Is Using Ultimate(Event Player)), Event Player.ab1Presses)));
		"S slot | K ability 1, ability 2, ultimate held | melee, reload held | R resource of jump, primary, secondary"
		Log To Inspector(Custom String("S{0} K {1} | {2}", Slot Of(Event Player), Custom String("{0} {1} {2}", Is Button Held(
			Event Player, Button(Ability 1)), Is Button Held(Event Player, Button(Ability 2)), Is Button Held(Event Player, Button(
			Ultimate))), Custom String("{0} {1} | R {2}", Is Button Held(Event Player, Button(Melee)), Is Button Held(Event Player, Button(
			Reload)), Custom String("{0} {1} {2}", Ability Resource(Event Player, Button(Jump)), Ability Resource(Event Player, Button(
			Primary Fire)), Ability Resource(Event Player, Button(Secondary Fire))))));
		"S slot | C resource of ability 1, ability 2 | Q charges of ability 1, ability 2 | CD cooldowns of ability 1, ability 2"
		Log To Inspector(Custom String("S{0} C {1} | {2}", Slot Of(Event Player), Custom String("{0} {1}", Ability Resource(Event Player,
			Button(Ability 1)), Ability Resource(Event Player, Button(Ability 2))), Custom String("Q {0} | CD {1}", Custom String(
			"{0} {1}", Ability Charge(Event Player, Button(Ability 1)), Ability Charge(Event Player, Button(Ability 2))), Custom String(
			"{0} {1}", Ability Cooldown(Event Player, Button(Ability 1)), Ability Cooldown(Event Player, Button(Ability 2))))));
		"S slot | X ultimate charge %, alternate form (D.Va's pilot) | W firing primary, secondary | H health, max | ST stunned, asleep, knocked down, hacked, phased out"
		Log To Inspector(Custom String("S{0} X {1} | {2}", Slot Of(Event Player), Custom String("{0} {1}", Ultimate Charge Percent(
			Event Player), Is In Alternate Form(Event Player)), Custom String("W {0} | H {1} | {2}", Custom String("{0} {1}",
			Is Firing Primary(Event Player), Is Firing Secondary(Event Player)), Custom String("{0} {1}", Health(Event Player),
			Max Health(Event Player)), Custom String("ST {0} {1} {2}", Has Status(Event Player, Stunned), Has Status(Event Player,
			Asleep), Custom String("{0} {1} {2}", Has Status(Event Player, Knocked Down), Has Status(Event Player, Hacked), Has Status(
			Event Player, Phased Out))))));
		Wait(0.016, Ignore Condition);
		Loop If Condition Is True;
	}
}

rule("Hero picked or changed: a bookmark")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Has Spawned(Event Player) == True;
		Is Dummy Bot(Event Player) == False;
		Hero Of(Event Player) != Event Player.hero;
	}

	actions
	{
		Event Player.hero = Hero Of(Event Player);
		Event Player.lastEvent = Custom String("HERO {0}", Hero Of(Event Player));
		Log To Inspector(Custom String("S{0} EVENT HERO | t {1} | {2}", Slot Of(Event Player), Total Time Elapsed, Hero Of(Event Player)));
	}
}

rule("On a wall (climbing or riding): a bookmark (no waits, so a quick next wall still gets its own)")

	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is On Wall(Event Player) == True;

	actions
	{
		Event Player.attachTime = Total Time Elapsed;

		Event Player.rideCount = Event Player.rideCount + 1;
		Event Player.lastEvent = Custom String("ATTACH {0} | vel {1} | input {2}", Event Player.rideCount, Velocity Of(Event Player),
			Throttle Of(Event Player));
		Log To Inspector(Custom String("S{0} EVENT ATTACH {1} | {2}", Slot Of(Event Player), Event Player.rideCount, Custom String(
			"t {0} | vel {1} | {2}", Total Time Elapsed, Velocity Of(Event Player), Custom String("pos {0} | face {1} | N {2}",
			Position Of(Event Player), Facing Direction Of(Event Player), Event Player.wallNormal))));
	}
}

rule("Off the wall: a bookmark (no waits)")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Is On Wall(Event Player) == False;
		Event Player.attachTime > Event Player.leaveTime;
		Is Dummy Bot(Event Player) == False;
	}

	actions
	{
		Event Player.leaveTime = Total Time Elapsed;
		Event Player.lastEvent = Custom String("LEAVE after {0} s | vel {1} | jump {2}", Round To Integer((Total Time Elapsed
			- Event Player.attachTime) * 100, Down) / 100, Velocity Of(Event Player), Is Button Held(Event Player, Button(Jump)));
		Log To Inspector(Custom String("S{0} EVENT LEAVE {1} | {2}", Slot Of(Event Player), Event Player.rideCount, Custom String(
			"t {0} | rode {1} s | {2}", Total Time Elapsed, Total Time Elapsed - Event Player.attachTime, Custom String(
			"vel {0} | rose {1} | jump {2}", Velocity Of(Event Player), Y Component Of(Position Of(Event Player)
			- Event Player.attachPos), Is Button Held(Event Player, Button(Jump))))));
	}
}
